// Menangani masalah CORS untuk Vercel & AI Studio (Preflight Request)
function doOptions(e) {
  return ContentService.createTextOutput("OK")
    .setMimeType(ContentService.MimeType.TEXT);
}

// Menangani request baca data (GET)
function doGet(e) {
  var action = e.parameter.action;
  
  if (action === 'init') {
    return ContentService.createTextOutput(JSON.stringify(getInitData()))
      .setMimeType(ContentService.MimeType.JSON);
  }
  
  if (action === 'cekToken') {
    return ContentService.createTextOutput(JSON.stringify(cekStatusToken(e.parameter.token)))
      .setMimeType(ContentService.MimeType.JSON);
  }
  
  if (action === 'getSoalMatDas') {
    return ContentService.createTextOutput(JSON.stringify(getSoalMatDas(e.parameter.paket)))
      .setMimeType(ContentService.MimeType.JSON);
  }
  
  return ContentService.createTextOutput(JSON.stringify({error: "Endpoint GET tidak valid"}))
    .setMimeType(ContentService.MimeType.JSON);
}

// Menangani request kirim/simpan data (POST)
function doPost(e) {
  try {
    var payload = JSON.parse(e.postData.contents);
    var action = payload.action;

    if (action === 'getSoalMatDas') {
      return ContentService.createTextOutput(JSON.stringify(getSoalMatDas(payload.paket)))
        .setMimeType(ContentService.MimeType.JSON);
    }

    if (action === 'verifikasi') {
      var result = verifikasiDanAmbilSoal(payload.token, payload.nama, payload.email, payload.posisi, payload.deviceId);
      return ContentService.createTextOutput(JSON.stringify(result))
        .setMimeType(ContentService.MimeType.JSON);
    }
    
    if (action === 'uploadSnapshot') {
      var resultSnapshot = uploadSnapshotBackground(payload.token, payload.nama, payload.stageKey, payload.fotoBase64);
      return ContentService.createTextOutput(JSON.stringify(resultSnapshot))
        .setMimeType(ContentService.MimeType.JSON);
    }

    if (action === 'simpanHasil') {
      var lock = LockService.getScriptLock();
      lock.waitLock(15000); // Tunggu antrean maksimal 15 detik
      
      try {
        var result = simpanHasilUjian(payload.dataKandidat);
        return ContentService.createTextOutput(JSON.stringify({status: result}))
          .setMimeType(ContentService.MimeType.JSON);
      } finally {
        lock.releaseLock();
      }
    }
    
    return ContentService.createTextOutput(JSON.stringify({error: "Endpoint POST tidak valid"}))
      .setMimeType(ContentService.MimeType.JSON);
      
  } catch(err) {
    return ContentService.createTextOutput(JSON.stringify({error: err.message}))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

function verifikasiDanAmbilSoal(tokenInput, nama, email, posisi, deviceId) {
  var ss = SpreadsheetApp.openById("1J-Kxp7bmKqiLkkwQnk4YWxvQj7YQeOrZmc2sXNRsulA"); // GANTI ID SPREADSHEET JIKA PERLU

  try {
    // === 1. CEK WAKTU ===
    var sheetPengaturan = ss.getSheetByName("PENGATURAN");
    var waktuSekarang = new Date();

    if (sheetPengaturan) {
      var waktuMulai   = sheetPengaturan.getRange("B3").getValue();
      var waktuSelesai = sheetPengaturan.getRange("B4").getValue();

      if (waktuMulai && waktuMulai.toString().trim() !== "") {
        var tMulai = new Date(waktuMulai);
        if (!isNaN(tMulai.getTime()) && waktuSekarang.getTime() < tMulai.getTime()) {
          return { error: "⏳ UJIAN BELUM DIMULAI!\n" + sheetPengaturan.getRange("B3").getDisplayValue() };
        }
      }
      if (waktuSelesai && waktuSelesai.toString().trim() !== "") {
        var tSelesai = new Date(waktuSelesai);
        if (!isNaN(tSelesai.getTime()) && waktuSekarang.getTime() > tSelesai.getTime()) {
          return { error: "⛔ AKSES DITUTUP!\n" + sheetPengaturan.getRange("B4").getDisplayValue() };
        }
      }
    }

    // === 2. CEK TOKEN ===
    var sheetToken = ss.getSheetByName("DATA_TOKEN");
    if (!sheetToken) return { error: "Sistem Error: Sheet DATA_TOKEN tidak ditemukan." };

    var dataToken = sheetToken.getDataRange().getValues();
    var barisDitemukan = -1;
    for (var i = 1; i < dataToken.length; i++) {
      if (dataToken[i][0].toString().trim() === tokenInput.trim()) {
        barisDitemukan = i;
        break;
      }
    }
    if (barisDitemukan === -1) return { error: "❌ TOKEN TIDAK TERDAFTAR!" };

    var statusToken = dataToken[barisDitemukan][1].toString().trim().toUpperCase();

    if (statusToken === "SELESAI") {
      return { error: "✅ Ujian Anda sudah tercatat dan selesai.\nTidak dapat mengakses ujian kembali." };
    }
    if (statusToken === "SUDAH DIPAKAI") {
      return { error: "❌ TOKEN HANGUS! Token ini sudah digunakan." };
    }

    var isRestore = (statusToken === "SEDANG DIKERJAKAN");

    // === PROTEKSI JOKI (CEK DEVICE ID) ===
    var savedDeviceId = dataToken[barisDitemukan][4] ? dataToken[barisDitemukan][4].toString().trim() : "";
    
    if (isRestore) {
      if (savedDeviceId !== "" && deviceId !== savedDeviceId) {
        return { error: "🚨 AKSES DITOLAK!\nToken ini sedang dikerjakan di perangkat/browser lain. Anda tidak bisa menggunakan lebih dari satu perangkat." };
      }
    }

    // === 3. SET STATUS TOKEN → SEDANG DIKERJAKAN ===
    var waktuAksesDetail = Utilities.formatDate(waktuSekarang, Session.getScriptTimeZone(), "dd/MM/yyyy HH:mm:ss");
    sheetToken.getRange(barisDitemukan + 1, 2).setValue("SEDANG DIKERJAKAN");
    sheetToken.getRange(barisDitemukan + 1, 3).setValue(nama + " (" + email + ")");
    
    if (!isRestore) {
      sheetToken.getRange(barisDitemukan + 1, 4).setValue(waktuAksesDetail);
      sheetToken.getRange(barisDitemukan + 1, 5).setValue(deviceId); 
    }

    // === 4. SEDOT SOAL MATEMATIKA DASAR (3 PAKET: Paket A, Paket B, Paket C) ===
    var sheetMatematika = ss.getSheetByName("SOAL_HITUNGAN") || ss.getSheetByName("SOAL_MATEMATIKA");
    var paketTerpilihNama = "Paket A";
    var soalMatematikaList = [];

    if (sheetMatematika) {
      var dataMtk = sheetMatematika.getDataRange().getValues();
      if (dataMtk.length > 1) {
        var paketTersedia = [];
        for (var i = 1; i < dataMtk.length; i++) {
          var namaPaket = dataMtk[i][0].toString().trim();
          if (namaPaket !== "" && paketTersedia.indexOf(namaPaket) === -1) {
            paketTersedia.push(namaPaket);
          }
        }
        if (paketTersedia.length > 0) {
          paketTerpilihNama = paketTersedia[Math.floor(Math.random() * paketTersedia.length)];
        }
        for (var i = 1; i < dataMtk.length; i++) {
          if (dataMtk[i][0].toString().trim() === paketTerpilihNama || paketTersedia.length === 0) {
            var opsi = [];
            for (var col = 2; col <= 6; col++) {
              var teksOpsi = dataMtk[i][col] ? dataMtk[i][col].toString().trim() : "";
              if (teksOpsi !== "" && teksOpsi !== "-") opsi.push(teksOpsi);
            }
            soalMatematikaList.push({
              id: "MTK_" + (i + 1),
              soal: dataMtk[i][1].toString(),
              opsi: opsi
            });
          }
        }
      }
    }

    return {
      success: true,
      isRestore: isRestore,
      matematika: soalMatematikaList,
      paketMatematika: paketTerpilihNama
    };

  } catch(e) {
    return { error: "💥 Terjadi kesalahan sistem: " + e.message };
  }
}

// =====================================================================
// FETCH SOAL MATEMATIKA DASAR SECARA DINAMIS (SHEET: SOAL_HITUNGAN)
// Kolom A: Paket Soal
// Kolom B: Teks Soal
// Kolom C, D, E, F, G: Pilihan Ganda (A, B, C, D, E)
// Kolom H: Kunci Jawaban (DISEMBUNYIKAN / TIDAK DIKIRIM KE FRONTEND)
// =====================================================================
function getSoalMatDas(paketInput) {
  var ss = SpreadsheetApp.openById("1J-Kxp7bmKqiLkkwQnk4YWxvQj7YQeOrZmc2sXNRsulA"); // GANTI ID SPREADSHEET JIKA PERLU
  var sheet = ss.getSheetByName("SOAL_HITUNGAN") || ss.getSheetByName("SOAL_MATEMATIKA");

  if (!sheet) {
    return {
      success: false,
      error: "Sheet SOAL_HITUNGAN tidak ditemukan.",
      soal: []
    };
  }

  var data = sheet.getDataRange().getValues();
  if (data.length <= 1) {
    return {
      success: false,
      error: "Data soal kosong pada sheet SOAL_HITUNGAN.",
      soal: []
    };
  }

  // 1. Identifikasi daftar paket unik yang tersedia
  var paketTersedia = [];
  for (var i = 1; i < data.length; i++) {
    var p = (data[i][0] || "").toString().trim();
    if (p !== "" && paketTersedia.indexOf(p) === -1) {
      paketTersedia.push(p);
    }
  }

  // 2. Tentukan paket yang dipilih (filter spesifik atau acak dari paket tersedia)
  var paketTerpilih = (paketInput && paketInput.toString().trim() !== "")
    ? paketInput.toString().trim()
    : (paketTersedia.length > 0 ? paketTersedia[Math.floor(Math.random() * paketTersedia.length)] : "Paket A");

  var listSoal = [];
  for (var r = 1; r < data.length; r++) {
    var rowPaket = (data[r][0] || "").toString().trim();

    if (rowPaket === paketTerpilih || paketTersedia.length === 0) {
      var teksSoal = (data[r][1] || "").toString().trim();
      if (teksSoal === "") continue;

      var opsiArr = [];
      for (var c = 2; c <= 6; c++) {
        var optVal = (data[r][c] !== undefined && data[r][c] !== null) ? data[r][c].toString().trim() : "";
        if (optVal !== "" && optVal !== "-") {
          opsiArr.push(optVal);
        }
      }

      // CATATAN KEAMANAN:
      // Kolom H (data[r][7]) adalah Kunci Jawaban.
      // Kunci INI SENGAJA TIDAK DIKIRIM ke frontend agar tidak dapat dibaca di DevTools browser!
      // Kunci tetap aman tersimpan di spreadsheet untuk divalidasi oleh fungsi simpanHasilUjian.
      listSoal.push({
        id: "MTK_" + (r + 1),
        paket: rowPaket || paketTerpilih,
        soal: teksSoal,
        opsi: opsiArr
      });
    }
  }

  return {
    success: true,
    paketTerpilih: paketTerpilih,
    paketTersedia: paketTersedia,
    soal: listSoal
  };
}

// KOREKSI RESMI SOAL ARMY ALPHA (12 NOMOR BERDASARKAN KUNCI PDF)
// =====================================================================
function koreksiArmyAlpha(alphaData) {
  if (!alphaData) {
    return { skor: 0, skorPersen: 0, log: "Tidak Dikerjakan" };
  }

  var benarCount = 0;
  var logList = [];

  function getInputs(soalId) {
    var item = alphaData[soalId];
    return item && item.inputan ? item.inputan.map(function(s) { return (s || "").toString().trim().toUpperCase(); }) : [];
  }
  function getCoret(soalId) {
    var item = alphaData[soalId];
    return item && item.dicoret ? item.dicoret.map(function(s) { return s.toString().trim(); }) : [];
  }
  function getGarisBawah(soalId) {
    var item = alphaData[soalId];
    return item && item.garisBawah ? item.garisBawah.map(function(s) { return s.toString().trim().toUpperCase(); }) : [];
  }
  function getIrisan(soalId) {
    var item = alphaData[soalId];
    return item && item.irisan ? item.irisan : {};
  }
  function getJalur(soalId) {
    var item = alphaData[soalId];
    return item && item.jalurTerpilih ? item.jalurTerpilih : "";
  }

  // Soal 1: Lingkaran ke-2 = X, Lingkaran ke-5 = huruf (bukan X dan tidak kosong)
  var inp1 = getInputs("ALP_1");
  var s1 = (inp1[1] === "X" && inp1[4] !== "X" && inp1[4] !== "" && inp1[4] !== undefined);
  if (s1) benarCount++;
  logList.push("N1:" + (s1 ? "B" : "S"));

  // Soal 2: Garis 3 ke 6 lewat bawah 4 dan atas 5
  var rute2 = getJalur("ALP_2");
  var s2 = (rute2 === "bawah4_atas5");
  if (s2) benarCount++;
  logList.push("N2:" + (s2 ? "B" : "S"));

  // Soal 3: Koma dalam lingkaran pertama & 'Tidak' tidak dicoret
  var inp3 = getInputs("ALP_3");
  var cor3 = getCoret("ALP_3");
  var s3 = (inp3[0] === "," || inp3[0] === ".") && (cor3.indexOf("Tidak") === -1 && cor3.indexOf("tidak") === -1);
  if (s3) benarCount++;
  logList.push("N3:" + (s3 ? "B" : "S"));

  // Soal 4: Lingkaran 3 = 7, Lingkaran 4 = kosong, Lingkaran 5 = jawaban salah (angka selain 7)
  var inp4 = getInputs("ALP_4");
  var s4 = (inp4[2] === "7" && (!inp4[3] || inp4[3] === "" || inp4[3] === "-") && inp4[4] !== "" && inp4[4] !== "7" && inp4[4] !== undefined);
  if (s4) benarCount++;
  logList.push("N4:" + (s4 ? "B" : "S"));

  // Soal 5: X di lingkaran luar segitiga, A di irisan segitiga-lingkaran
  var ir5 = getIrisan("ALP_5");
  var s5 = ((ir5.lingkaran_luar || "").toUpperCase() === "X" && (ir5.irisan || "").toUpperCase() === "A");
  if (s5) benarCount++;
  logList.push("N5:" + (s5 ? "B" : "S"));

  // Soal 6: X dalam segitiga luar bujur sangkar, Y dalam bujur sangkar-segitiga luar lingkaran
  var ir6 = getIrisan("ALP_6");
  var s6 = ((ir6.segitiga_non_kotak || "").toUpperCase() === "X" && (ir6.kotak_segitiga_non_lingkaran || "").toUpperCase() === "Y");
  if (s6) benarCount++;
  logList.push("N6:" + (s6 ? "B" : "S"));

  // Soal 7: Garis bawah di D, garis melintang di K
  var gb7 = getGarisBawah("ALP_7");
  var cor7 = getCoret("ALP_7");
  var s7 = (gb7.indexOf("D") !== -1 && cor7.indexOf("K") !== -1);
  if (s7) benarCount++;
  logList.push("N7:" + (s7 ? "B" : "S"));

  // Soal 8: Lingkaran 1 = A, Lingkaran 2 = A, Lingkaran 3 = G
  var inp8 = getInputs("ALP_8");
  var s8 = (inp8[0] === "A" && inp8[1] === "A" && inp8[2] === "G");
  if (s8) benarCount++;
  logList.push("N8:" + (s8 ? "B" : "S"));

  // Soal 9: Coret bilangan 33 dan 36
  var cor9 = getCoret("ALP_9");
  var s9 = (cor9.indexOf("33") !== -1 && cor9.indexOf("36") !== -1 && cor9.length === 2);
  if (s9) benarCount++;
  logList.push("N9:" + (s9 ? "B" : "S"));

  // Soal 10: Petak kecil (idx 0 & 3) = 8 atau 6, petak hampir paling besar (idx 2) = angka 1..5
  var inp10 = getInputs("ALP_10");
  var s10 = ((inp10[0] === "8" || inp10[0] === "6") &&
             (inp10[3] === "8" || inp10[3] === "6") &&
             (["1", "2", "3", "4", "5"].indexOf(inp10[2]) !== -1));
  if (s10) benarCount++;
  logList.push("N10:" + (s10 ? "B" : "S"));

  // Soal 11: Coret ganjil tanpa huruf luar segitiga [5, 9, 3] & genap berhuruf dalam lingkaran [6A]
  var cor11 = getCoret("ALP_11");
  var c0 = cor11.indexOf(0) !== -1 || cor11.indexOf("0") !== -1;
  var c1 = cor11.indexOf(1) !== -1 || cor11.indexOf("1") !== -1;
  var c3 = cor11.indexOf(3) !== -1 || cor11.indexOf("3") !== -1;
  var c7 = cor11.indexOf(7) !== -1 || cor11.indexOf("7") !== -1;
  var s11 = (c0 && c1 && c3 && c7 && cor11.length === 4);
  if (s11) benarCount++;
  logList.push("N11:" + (s11 ? "B" : "S"));

  // Soal 12: Coret angka 4
  var cor12 = getCoret("ALP_12");
  var s12 = (cor12.indexOf("4") !== -1 && cor12.length === 1);
  if (s12) benarCount++;
  logList.push("N12:" + (s12 ? "B" : "S"));

  var persentaseAlpha = Math.round((benarCount / 12) * 100);
  return {
    skor: benarCount,
    skorPersen: persentaseAlpha,
    log: logList.join(" | ")
  };
}

  }

  var benar = 0;

      totalSoal++;

      if (kunciJawaban !== "" && jawabanUser === kunciJawaban) {
        benar++;
        detailLog.push(key + ":B");
      } else {
        detailLog.push(key + ":S(" + (jawabanUser === "-" || !jawabanUser ? "Kosong" : jawabanUser) + ")");
      }
    }
  }

  return {
    skor: benar,
    total: totalSoal,
    log: detailLog.join(" | ")
  };
}

// FUNGSI INTERPRETASI SKOR PSIKOTES (ALPHA, MATDAS, RAC)
// =====================================================================
function getInterpretasiAlpha(skor) {
  var s = Number(skor) || 0;
  if (s <= 0) return "Kurang Sekali";
  if (s <= 2) return "Kurang";
  if (s <= 5) return "Cukup";
  if (s <= 8) return "Baik";
  return "Baik Sekali"; // > 8
}

function getInterpretasiMatDas(skor) {
  var s = Number(skor) || 0;
  if (s <= 4) return "Kurang Sekali";
  if (s <= 8) return "Kurang";
  if (s <= 12) return "Cukup";
  if (s <= 16) return "Baik";
  return "Baik Sekali"; // > 16
}

function getInterpretasiRAC(skor) {
  var s = Number(skor) || 0;
  if (s <= 15) return "Kurang Sekali";
  if (s <= 21) return "Kurang";
  if (s <= 25) return "Cukup";
  if (s <= 31) return "Baik";
  return "Baik Sekali"; // > 31
}

function isMinimalCukup(interpretasi) {
  return interpretasi === "Cukup" || interpretasi === "Baik" || interpretasi === "Baik Sekali";
}

function hitungHasilAkhir(interpAlpha, interpMatDas, interpRAC, isDiskualifikasi) {
  if (isDiskualifikasi) {
    return "Tidak Lulus";
  }
  var alphaCukup = isMinimalCukup(interpAlpha);
  var matDasCukup = isMinimalCukup(interpMatDas);
  var racCukup = isMinimalCukup(interpRAC);

  if (alphaCukup && matDasCukup && racCukup) {
    return "Lulus";
  }
  if (alphaCukup && racCukup && !matDasCukup) {
    return "Dipertimbangkan untuk Helper";
  }
  return "Tidak Lulus";
}

function getOrCreateCandidateFolder(namaKandidat, token) {

// 1. ASYNC UPLOAD (PENGIRIMAN BACKGROUND):
// Menerima file snapshot proctoring secara fire-and-forget tanpa memblokir ujian kandidat
function uploadSnapshotBackground(token, nama, stageKey, fotoBase64) {
  try {
    var subFolder = getOrCreateCandidateFolder(nama, token);
    if (!subFolder) {
      return { success: false, error: "Gagal membuat/mengakses folder Drive kandidat." };
    }

    var cleanStage = (stageKey || "snapshot").replace(/[^a-zA-Z0-9_-]/g, "_");

    // 2. GRACEFUL FALLBACK UNTUK KAMERA MATI (KHUSUS IOS/SAFARI):
    // Jika kamera diputus/dimatikan oleh OS Safari, buat file catatan peringatan HRD
    if (!fotoBase64 || fotoBase64 === "CAMERA_BLOCKED_OR_ERROR" || typeof fotoBase64 !== "string") {
      var timeStr = Utilities.formatDate(new Date(), Session.getScriptTimeZone(), "dd/MM/yyyy HH:mm:ss");
      var logMsg = "WAKTU: " + timeStr + "\n" +
                   "TAHAP: " + cleanStage + "\n" +
                   "STATUS: CAMERA_BLOCKED_OR_ERROR\n" +
                   "CATATAN: Stream kamera mati, stream terputus, atau izin dimatikan oleh sistem operasi perangkat (iOS/Safari) saat transisi ujian.";
      subFolder.createFile("LOG_WARNING_" + cleanStage + ".txt", logMsg, "text/plain");
      return { success: true, status: "logged_camera_error", folderUrl: subFolder.getUrl() };
    }

    // Jika foto valid berformat Base64
    if (fotoBase64.indexOf("base64,") !== -1) {
      var cleanImg = fotoBase64.split("base64,")[1];
      var decodedImg = Utilities.base64Decode(cleanImg);
      var fileName = cleanStage + ".jpg";
      var blobImg = Utilities.newBlob(decodedImg, "image/jpeg", fileName);
      subFolder.createFile(blobImg);
      return { success: true, fileName: fileName, folderUrl: subFolder.getUrl() };
    }

    return { success: false, error: "Format payload foto tidak valid" };
  } catch (err) {
    Logger.log("Error uploadSnapshotBackground: " + err.message);
    return { success: false, error: err.message };
  }
}

function simpanFotoKeDrive(folderIndukId, namaKandidat, token, fotoSelfie, fotoProctoring) {
  try {
    var subFolder = getOrCreateCandidateFolder(namaKandidat, token);
    if (!subFolder) return "-";

    // 1. Simpan Foto Selfie Master Awal jika belum diunggah secara async
    if (fotoSelfie && typeof fotoSelfie === "string" && fotoSelfie.indexOf("base64,") !== -1) {
      try {
        var cleanSelfie = fotoSelfie.split("base64,")[1];
        var decodedSelfie = Utilities.base64Decode(cleanSelfie);
        var blobSelfie = Utilities.newBlob(decodedSelfie, "image/jpeg", "01_Selfie_Awal.jpg");
        subFolder.createFile(blobSelfie);
      } catch (errSelfie) {}
    }

    // 2. Simpan Foto-Foto Snapshot Proctoring jika ada dalam payload cadangan
    if (fotoProctoring && typeof fotoProctoring === "object") {
      var stepIdx = 2;
      for (var key in fotoProctoring) {
        if (fotoProctoring.hasOwnProperty(key)) {
          var dataFoto = fotoProctoring[key];
          if (dataFoto === "CAMERA_BLOCKED_OR_ERROR") {
            try {
              subFolder.createFile("LOG_WARNING_" + key + ".txt", "Kamera terputus atau dimatikan oleh iOS pada " + key, "text/plain");
            } catch (e) {}
          } else if (dataFoto && typeof dataFoto === "string" && dataFoto.indexOf("base64,") !== -1) {
            try {
              var cleanImg = dataFoto.split("base64,")[1];
              var decodedImg = Utilities.base64Decode(cleanImg);
              var fileName = "0" + stepIdx + "_" + key.replace(/[^a-zA-Z0-9_-]/g, "_") + ".jpg";
              var blobImg = Utilities.newBlob(decodedImg, "image/jpeg", fileName);
              subFolder.createFile(blobImg);
              stepIdx++;
            } catch (errSnap) {}
          }
        }
      }
    }

    return subFolder.getUrl();
  } catch (e) {
    return "Error Drive: " + e.message;
  }
}

function simpanHasilUjian(dataKandidat) {

    "Skor MatDas",
    "Interpretasi MatDas",
    "Skor RAC",
    "Interpretasi RAC",
    "HASIL AKHIR",
    "Status Pelanggaran",
    "Link Foto Proctoring",
    "Log Detail Jawaban"
  ];

    sheetHasil.getRange("A1:N1").setFontWeight("bold").setBackground("#0056b3").setFontColor("white");
    sheetHasil.setFrozenRows(1);
  } else {
    var lastCol = sheetHasil.getLastColumn();
    var firstRow = sheetHasil.getRange(1, 1, 1, Math.max(lastCol, headerLengkap.length)).getValues()[0];
    if (lastCol !== headerLengkap.length || firstRow[12] !== "Link Foto Proctoring") {
      sheetHasil.getRange(1, 1, 1, headerLengkap.length).setValues([headerLengkap]);
      sheetHasil.getRange(1, 1, 1, headerLengkap.length).setFontWeight("bold").setBackground("#0056b3").setFontColor("white");
    }
  }

  var sheetMatematika = ss.getSheetByName("SOAL_HITUNGAN") || ss.getSheetByName("SOAL_MATEMATIKA");
  var dataSheetMatematika = sheetMatematika ? sheetMatematika.getDataRange().getValues() : [];

  // 2. Hitung Skor & Log Masing-Masing Subtes
  // --- Tahap 1: Army Alpha ---
  var skorAlpha = 0;
  var logAlpha = "-";
  if (dataKandidat.skorAlpha !== undefined && dataKandidat.skorAlpha !== null) {
    skorAlpha = Number(dataKandidat.skorAlpha) || 0;
    logAlpha = "Skor: " + skorAlpha + "/12";
  } else {
    var resAlpha = koreksiArmyAlpha(dataKandidat.alpha);
    skorAlpha = resAlpha.skor;
    logAlpha = resAlpha.log;
  }

  // --- Tahap 2: Matematika Dasar ---
  var skorMatDas = 0;
  var logMatDas = "-";
  if (dataKandidat.skorMatDas !== undefined && dataKandidat.skorMatDas !== null) {
    skorMatDas = Number(dataKandidat.skorMatDas) || 0;
    logMatDas = "Skor: " + skorMatDas;
  } else {
    var resMatematika = koreksiMatematika(dataKandidat.matematika, dataSheetMatematika);
    skorMatDas = resMatematika.skor;
    logMatDas = resMatematika.log;
  }

  // --- Tahap 3: RAC ---
  var skorRAC = 0;
  var logRAC = "-";
  if (dataKandidat.skorRAC !== undefined && dataKandidat.skorRAC !== null) {
    skorRAC = Number(dataKandidat.skorRAC) || 0;
    logRAC = "Skor: " + skorRAC + "/40";
  } else {
    var resRAC = koreksiRAC(dataKandidat.rac);
    skorRAC = resRAC.skor;
    logRAC = resRAC.log;
  }

  // 3. Tentukan Interpretasi untuk Setiap Subtes
  var interpAlpha = getInterpretasiAlpha(skorAlpha);
  var interpMatDas = getInterpretasiMatDas(skorMatDas);
  var interpRAC = getInterpretasiRAC(skorRAC);

  var isDiskualifikasi = (dataKandidat.pelanggaran >= 3);
  var statusPelanggaran = isDiskualifikasi
    ? "DISKUALIFIKASI (" + dataKandidat.pelanggaran + "x)"
    : (dataKandidat.pelanggaran || 0) + " Peringatan";

  // 5. Penentuan HASIL AKHIR
  var hasilAkhir = hitungHasilAkhir(interpAlpha, interpMatDas, interpRAC, isDiskualifikasi);

  // 6. Simpan Foto Proctoring ke Google Drive Folder
  var linkFotoDrive = simpanFotoKeDrive(
    FOLDER_DRIVE_PROCTORING_ID,
    dataKandidat.nama,
    dataKandidat.token,
    dataKandidat.fotoSelfie,
    dataKandidat.fotoProctoring
  );

  var logDetailJawaban = "[ALPHA] " + logAlpha + " | [MATDAS] " + logMatDas + " | [RAC] " + logRAC;

    waktuSubmitDetail,
    dataKandidat.nama,
    dataKandidat.email,
    dataKandidat.posisi,
    skorAlpha,
    interpAlpha,
    skorMatDas,
    interpMatDas,
    skorRAC,
    interpRAC,
    hasilAkhir,
    statusPelanggaran,
    linkFotoDrive,
    logDetailJawaban
  ]);

  // 9. Update Status Token -> SELESAI
  var sheetToken = ss.getSheetByName("DATA_TOKEN");
  if (sheetToken) {
    var dataToken = sheetToken.getDataRange().getValues();
    for (var i = 1; i < dataToken.length; i++) {
      if (dataToken[i][0].toString().trim() === dataKandidat.token) {
        sheetToken.getRange(i + 1, 2).setValue("SELESAI");
        break;
      }
    }
  }

  return "Sukses";
}
